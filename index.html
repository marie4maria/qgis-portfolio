road_network_style.py — PyQGIS automation script
# Road Network Classifier — PyQGIS
# Applies categorized symbology by road_class field

from qgis.core import *
from PyQt5.QtGui import QColor

layer = QgsProject.instance().mapLayersByName('road_network')[0]

road_styles = {
    'highway':   {'color': '#c4410c', 'width': 1.2},
    'arterial':  {'color': '#d4a017', 'width': 0.8},
    'collector': {'color': '#1a6b5a', 'width': 0.6},
    'local':     {'color': '#8a7d6b', 'width': 0.4},
}

categories = []
for road_type, style in road_styles.items():
    sym = QgsSymbol.defaultSymbol(layer.geometryType())
    sym.setColor(QColor(style['color']))
    sym.setWidth(style['width'])
    cat = QgsRendererCategory(road_type, sym, road_type.title())
    categories.append(cat)

renderer = QgsCategorizedSymbolRenderer('road_class', categories)
layer.setRenderer(renderer)
layer.triggerRepaint()

# Batch export to multiple formats
formats = ['GPKG', 'ESRI Shapefile', 'GeoJSON']
for fmt in formats:
    QgsVectorFileWriter.writeAsVectorFormat(
        layer, f'/output/road_network_{fmt.lower()}', 'UTF-8',
        layer.crs(), fmt
    )
    print(f'✓ Exported: {fmt}')
